100template <typename CommType>
101SensorEvent BNO085<CommType>::GetLatest(BNO085Sensor sensor) const {
102 SensorEvent out{};
103 out.sensor = sensor;
104 auto id = static_cast<uint8_t>(sensor);
105 const auto& val = latest_[id];
106 out.timestamp = val.timestamp;
107 uint8_t accuracy = val.status & 0x03;
108 switch (sensor) {
109 case BNO085Sensor::Accelerometer:
110 case BNO085Sensor::LinearAcceleration:
111 case BNO085Sensor::Gravity:
112 out.vector.x = val.un.accelerometer.x;
113 out.vector.y = val.un.accelerometer.y;
114 out.vector.z = val.un.accelerometer.z;
115 out.vector.accuracy = accuracy;
116 break;
117 case BNO085Sensor::Gyroscope:
118 out.vector.x = val.un.gyroscope.x;
119 out.vector.y = val.un.gyroscope.y;
120 out.vector.z = val.un.gyroscope.z;
121 out.vector.accuracy = accuracy;
122 break;
123 case BNO085Sensor::Magnetometer:
124 out.vector.x = val.un.magneticField.x;
125 out.vector.y = val.un.magneticField.y;
126 out.vector.z = val.un.magneticField.z;
127 out.vector.accuracy = accuracy;
128 break;
129 case BNO085Sensor::RotationVector:
130 case BNO085Sensor::GameRotationVector:
131 case BNO085Sensor::GeomagneticRotationVector:
132 case BNO085Sensor::ARVRStabilizedRV:
133 case BNO085Sensor::ARVRStabilizedGameRV:
134 out.rotation.w = val.un.rotationVector.real;
135 out.rotation.x = val.un.rotationVector.i;
136 out.rotation.y = val.un.rotationVector.j;
137 out.rotation.z = val.un.rotationVector.k;
138 out.rotation.accuracy = accuracy;
139 break;
140 case BNO085Sensor::GyroIntegratedRV:
141 out.rotation.w = val.un.gyroIntegratedRV.real;
142 out.rotation.x = val.un.gyroIntegratedRV.i;
143 out.rotation.y = val.un.gyroIntegratedRV.j;
144 out.rotation.z = val.un.gyroIntegratedRV.k;
145 break;
146 case BNO085Sensor::StepCounter:
147 out.stepCount = val.un.stepCounter.steps;
148 break;
149 case BNO085Sensor::TapDetector:
150 out.tap.doubleTap = (val.un.tapDetector.flags & TAPDET_DOUBLE);
151 if (val.un.tapDetector.flags & TAPDET_X) {
152 out.tap.direction = (val.un.tapDetector.flags & TAPDET_X_POS) ? 0 : 1;
153 } else if (val.un.tapDetector.flags & TAPDET_Y) {
154 out.tap.direction = (val.un.tapDetector.flags & TAPDET_Y_POS) ? 2 : 3;
155 } else if (val.un.tapDetector.flags & TAPDET_Z) {
156 out.tap.direction = (val.un.tapDetector.flags & TAPDET_Z_POS) ? 4 : 5;
157 } else {
158 out.tap.direction = 0;
159 }
160 out.detected = val.un.tapDetector.flags & (TAPDET_X | TAPDET_Y | TAPDET_Z);
161 break;
162 default:
163 break;
164 }
165 return out;
166}

180 // Store the HAL pointer in the RvcHalC_t adapter context
181 rvc_hal_adapter_.ctx = &hal;
182
183 // Set up C function-pointer trampolines that call through to the CRTP HAL
184 rvc_hal_adapter_.open = [](void* ctx) -> int {

187 rvc_hal_adapter_.close = [](void* ctx) {

263void BNO085<CommType>::handleSensorEvent(const sh2_SensorEvent_t* event) noexcept {
264 sh2_SensorValue_t value;
265 sh2_decodeSensorEvent(&value, event);
266 uint8_t id = value.sensorId;
267 if (id >= latest_.size())
268 return;
269 latest_[id] = value;
270 new_flag_[id] = true;
271 if (callback_) {
272 callback_(GetLatest(static_cast<BNO085Sensor>(id)));
273 new_flag_[id] = false;
