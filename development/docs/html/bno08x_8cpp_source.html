128template <typename CommType>
129SensorEvent BNO085<CommType>::GetLatest(BNO085Sensor sensor) const {
130 SensorEvent out{};
131 out.sensor = sensor;
132 auto id = static_cast<uint8_t>(sensor);
133 const auto& val = latest_[id];
134 out.timestamp = val.timestamp;
135 uint8_t accuracy = val.status & 0x03;
136 switch (sensor) {
137 case BNO085Sensor::Accelerometer:
138 case BNO085Sensor::LinearAcceleration:
139 case BNO085Sensor::Gravity:
140 out.vector.x = val.un.accelerometer.x;
141 out.vector.y = val.un.accelerometer.y;
142 out.vector.z = val.un.accelerometer.z;
143 out.vector.accuracy = accuracy;
144 break;
145 case BNO085Sensor::Gyroscope:
146 out.vector.x = val.un.gyroscope.x;
147 out.vector.y = val.un.gyroscope.y;
148 out.vector.z = val.un.gyroscope.z;
149 out.vector.accuracy = accuracy;
150 break;
151 case BNO085Sensor::Magnetometer:
152 out.vector.x = val.un.magneticField.x;
153 out.vector.y = val.un.magneticField.y;
154 out.vector.z = val.un.magneticField.z;
155 out.vector.accuracy = accuracy;
156 break;
157 case BNO085Sensor::RotationVector:
158 case BNO085Sensor::GameRotationVector:
159 case BNO085Sensor::GeomagneticRotationVector:
160 case BNO085Sensor::ARVRStabilizedRV:
161 case BNO085Sensor::ARVRStabilizedGameRV:
162 out.rotation.w = val.un.rotationVector.real;
163 out.rotation.x = val.un.rotationVector.i;
164 out.rotation.y = val.un.rotationVector.j;
165 out.rotation.z = val.un.rotationVector.k;
166 out.rotation.accuracy = accuracy;
167 break;
168 case BNO085Sensor::GyroIntegratedRV:
169 out.rotation.w = val.un.gyroIntegratedRV.real;
170 out.rotation.x = val.un.gyroIntegratedRV.i;
171 out.rotation.y = val.un.gyroIntegratedRV.j;
172 out.rotation.z = val.un.gyroIntegratedRV.k;
173 break;
174 case BNO085Sensor::StepCounter:
175 out.stepCount = val.un.stepCounter.steps;
176 break;
177 case BNO085Sensor::TapDetector:
178 out.tap.doubleTap = (val.un.tapDetector.flags & TAPDET_DOUBLE);
179 if (val.un.tapDetector.flags & TAPDET_X)
180 out.tap.direction = (val.un.tapDetector.flags & TAPDET_X_POS) ? 0 : 1;
181 else if (val.un.tapDetector.flags & TAPDET_Y)
182 out.tap.direction = (val.un.tapDetector.flags & TAPDET_Y_POS) ? 2 : 3;
183 else if (val.un.tapDetector.flags & TAPDET_Z)
184 out.tap.direction = (val.un.tapDetector.flags & TAPDET_Z_POS) ? 4 : 5;
185 else
186 out.tap.direction = 0;
187 out.detected = val.un.tapDetector.flags & (TAPDET_X | TAPDET_Y | TAPDET_Z);
188 break;
189 default:
190 break;
191 }
192 return out;
193}

433template <typename CommType>
434void BNO085<CommType>::rvcProcessByte(uint8_t c) noexcept {
435 if (rvc_frame_len_ == RVC_FRAME_LEN_) {
436 std::memmove(rvc_frame_, rvc_frame_ + 1, RVC_FRAME_LEN_ - 1);
437 rvc_frame_[RVC_FRAME_LEN_ - 1] = c;
438 } else {
439 rvc_frame_[rvc_frame_len_++] = c;
440 }
441
442 if (rvc_frame_len_ == RVC_FRAME_LEN_ &&
443 rvc_frame_[0] == 0xAA && rvc_frame_[1] == 0xAA) {
444 uint8_t check = 0;
445 for (int i = 2; i < RVC_FRAME_LEN_ - 1; ++i)
446 check += rvc_frame_[i];
447
448 if (check == rvc_frame_[RVC_FRAME_LEN_ - 1]) {
449 RvcSensorEvent event{};
450 event.timestamp_uS = io_.GetTimeUs();
451 event.index = rvc_frame_[2];
452 event.yaw = static_cast<int16_t>((rvc_frame_[4] << 8) | rvc_frame_[3]);
453 event.pitch = static_cast<int16_t>((rvc_frame_[6] << 8) | rvc_frame_[5]);
454 event.roll = static_cast<int16_t>((rvc_frame_[8] << 8) | rvc_frame_[7]);
455 event.acc_x = static_cast<int16_t>((rvc_frame_[10] << 8) | rvc_frame_[9]);
456 event.acc_y = static_cast<int16_t>((rvc_frame_[12] << 8) | rvc_frame_[11]);
457 event.acc_z = static_cast<int16_t>((rvc_frame_[14] << 8) | rvc_frame_[13]);
458 event.mi = rvc_frame_[15];
459 event.mr = rvc_frame_[16];
460
461 if (rvc_cb_) {
462 RvcSensorValue val;
463 decodeRvc(&val, &event);
464 rvc_cb_(val);
465 }
466 rvc_frame_len_ = 0;
467 }
468 }

525template <typename CommType>
526void BNO085<CommType>::dfuAppendCrc(uint8_t* packet, uint8_t len) noexcept {
527 uint16_t crc = 0xFFFF;
528 for (int n = 0; n < len; n++) {
529 uint16_t x = static_cast<uint16_t>(packet[n]) << 8;
530 for (int i = 0; i < 8; i++) {
531 if ((crc ^ x) & 0x8000) crc = (crc << 1) ^ 0x1021;
532 else crc = crc << 1;
533 x <<= 1;
534 }
535 }
536 packet[len] = (crc >> 8) & 0xFF;
537 packet[len + 1] = crc & 0xFF;
538}

548template <typename CommType>
549int BNO085<CommType>::dfuSend(uint8_t* dfuBuff, uint8_t* pData, uint32_t len) noexcept {
550 unsigned int retries = 0;
551 int status = SH2_OK;
552 uint8_t ack = 0;
553 bool gotAck = false;
554 uint32_t t;
555 sh2_Hal_t* hal = halWrapper_.asHal();
556
557 while (!gotAck && (retries < DFU_MAX_ATTEMPTS)) {
558 uint32_t now = hal->getTimeUs(hal);
559 uint32_t start = now;
560 status = 0;
561 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
562 status = hal->write(hal, pData, len);
563 now = hal->getTimeUs(hal);
564 }
565 if (status == 0) status = SH2_ERR_TIMEOUT;
566 if (status > 0) {
567 status = 0;
568 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
569 status = hal->read(hal, &ack, 1, &t);
570 now = hal->getTimeUs(hal);
571 }
572 if (status == 0) status = SH2_ERR_TIMEOUT;
573 }
574 if (status > 0) {
575 if (ack == DFU_ACK) { gotAck = true; status = SH2_OK; }
576 else { gotAck = false; status = SH2_ERR_HUB; }
577 }
578 if (!gotAck) retries++;
579 }
580 return (status >= 0) ? SH2_OK : status;
581}

619template <typename CommType>
620int BNO085<CommType>::Dfu(const HcBin_t& fw) noexcept {
621 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC)
622 return SH2_ERR;
623
624 int rc, status = SH2_OK;
625 uint32_t appLen = 0;
626 uint8_t packetLen = 0;
627 uint32_t offset = 0;
628 const char* s = nullptr;
629 uint8_t dfuBuff[DFU_MAX_PACKET_LEN + 2];
630 sh2_Hal_t* hal = halWrapper_.asHal();
631
632 rc = fw.open();
633 if (rc != 0) { status = SH2_ERR; goto dfu_end; }
634
635 s = fw.getMeta("FW-Format");
636 if (!s || std::strcmp(s, "BNO_V1") != 0) { status = SH2_ERR_BAD_PARAM; goto dfu_close; }
637
638 s = fw.getMeta("SW-Part-Number");
639 if (!s) { status = SH2_ERR_BAD_PARAM; goto dfu_close; }
640 if (std::strcmp(s, "1000-3608") != 0 && std::strcmp(s, "1000-3676") != 0 &&
641 std::strcmp(s, "1000-4148") != 0 && std::strcmp(s, "1000-4563") != 0) {
642 status = SH2_ERR_BAD_PARAM; goto dfu_close;
643 }
644
645 appLen = fw.getAppLen();
646 if (appLen < 1024) { status = SH2_ERR_BAD_PARAM; goto dfu_close; }
647
648 packetLen = fw.getPacketLen();
649 if (packetLen == 0 || packetLen > DFU_MAX_PACKET_LEN) packetLen = DFU_MAX_PACKET_LEN;
650
651 status = hal->open(hal);
652 if (status != SH2_OK) goto dfu_close;
653
654 status = dfuSendAppSize(dfuBuff, appLen);
655 if (status != SH2_OK) goto dfu_close;
656 status = dfuSendPktSize(dfuBuff, packetLen);
657 if (status != SH2_OK) goto dfu_close;
658
659 offset = 0;
660 while (offset < appLen) {
661 uint32_t toSend = appLen - offset;
662 if (toSend > packetLen) toSend = packetLen;
663 status = fw.getAppData(dfuBuff, offset, toSend);
664 if (status != SH2_OK) goto dfu_close;
665 status = dfuSendPkt(dfuBuff, dfuBuff, toSend);
666 if (status != SH2_OK) goto dfu_close;
667 offset += toSend;
668 }
669
670dfu_close:
671 fw.close();
672 if (status == SH2_OK) {
673 uint32_t now = hal->getTimeUs(hal), start = now;
674 while ((now - start) < DFU_DELAY_POST_US) now = hal->getTimeUs(hal);
675 }
676 hal->close(hal);
677
678dfu_end:
679 return status;
680}
