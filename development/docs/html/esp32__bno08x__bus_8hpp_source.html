166 int Read(uint8_t* data, uint32_t length) noexcept {
167 if (!initialized_ || dev_handle_ == nullptr) {
168 ESP_LOGE(TAG_I2C, "I2C bus not initialized");
169 return -1;
170 }
171
172 // If interrupt pin is configured, check it first to avoid unnecessary I2C transactions
173 if (config_.int_pin != GPIO_NUM_NC && !DataAvailable()) {
174 // Interrupt pin indicates no data available (pin is high, active low)
175 return 0; // Return 0 to indicate no data (SH-2 library expects this)
176 }
177
178 esp_err_t ret = i2c_master_receive(dev_handle_, data, length, pdMS_TO_TICKS(1000));
179 if (ret != ESP_OK) {
180 // ESP_ERR_TIMEOUT can occur if device is clock stretching (no data yet)
181 // This is acceptable for BNO08x - return 0 to indicate no data available
182 if (ret == ESP_ERR_TIMEOUT) {
183 return 0;
184 }
185 ESP_LOGE(TAG_I2C, "I2C read failed: %s (addr=0x%02X, len=%lu)",
186 esp_err_to_name(ret), config_.device_address,
187 static_cast<unsigned long>(length));
188 return -1;
189 }
190
191 return static_cast<int>(length);
192 }

205 bool Probe() noexcept {
206 if (!initialized_ || dev_handle_ == nullptr) {
207 return false;
208 }
209
210 // Try to read 1 byte with longer timeout to allow for clock stretching
211 // BNO08x will stretch clock if no data (timeout is OK - means device is present)
212 // NAK means device not present
213 uint8_t dummy;
214 esp_err_t ret = i2c_master_receive(dev_handle_, &dummy, 1, pdMS_TO_TICKS(100));
215
216 // Log the actual error for debugging
217 if (ret != ESP_OK && ret != ESP_ERR_TIMEOUT) {
218 ESP_LOGW(TAG_I2C, "Probe at 0x%02X returned: %s (0x%x)",
219 config_.device_address, esp_err_to_name(ret), ret);
220 }
221
222 // ESP_OK = data received (ACK + data) - device present and responding
223 // ESP_ERR_TIMEOUT = clock stretching exceeded timeout (device present, no data yet - OK for BNO08x)
224 // ESP_ERR_INVALID_RESPONSE = NAK during data phase (but ACK on address means device is present!)
225 // For BNO08x, if we get ACK on address (which we see on logic analyzer),
226 // the device IS present, even if it NAKs during data phase (no data ready)
227 // So we accept timeout and also check if it's a response error (which might indicate ACK then NAK)
228 if (ret == ESP_OK || ret == ESP_ERR_TIMEOUT) {
229 return true;
230 }
231
232 // ESP_ERR_INVALID_RESPONSE typically means NAK during data transfer
233 // But if address was ACKed (seen on logic analyzer), device is present
234 // BNO08x might NAK if it has no data ready, so we should still consider it present
235 if (ret == ESP_ERR_INVALID_RESPONSE) {
236 ESP_LOGI(TAG_I2C, "Probe got NAK during data phase, but address was ACKed - device present");
237 return true; // Device ACKed address, so it's present
238 }
239
240 return false; // Other errors indicate real problems
241 }

330 bool Init() noexcept {
331 if (initialized_) {
332 ESP_LOGW(TAG_I2C, "I2C bus already initialized");
333 return true;
334 }
335
336 ESP_LOGI(TAG_I2C, "Initializing I2C bus on port %d (SDA:GPIO%d, SCL:GPIO%d, "
337 "Freq:%lu Hz, Addr:0x%02X)",
338 config_.port, config_.sda_pin, config_.scl_pin,
339 static_cast<unsigned long>(config_.frequency), config_.device_address);
340
341 // Step 1: Configure GPIO pins (interrupt, reset)
342 if (!initGPIO()) {
343 ESP_LOGE(TAG_I2C, "Failed to initialize GPIO pins");
344 return false;
345 }
346
347 // Step 2: Create I2C master bus
348 i2c_master_bus_config_t bus_config = {};
349 bus_config.i2c_port = config_.port;
350 bus_config.sda_io_num = config_.sda_pin;
351 bus_config.scl_io_num = config_.scl_pin;
352 bus_config.clk_source = I2C_CLK_SRC_DEFAULT;
353 bus_config.glitch_ignore_cnt = 7;
354 bus_config.flags.enable_internal_pullup = config_.pullup_enable;
355 bus_config.intr_priority = 0;
356 bus_config.trans_queue_depth = 0;
357
358 esp_err_t ret = i2c_new_master_bus(&bus_config, &bus_handle_);
359 if (ret != ESP_OK) {
360 ESP_LOGE(TAG_I2C, "Failed to create I2C master bus: %s", esp_err_to_name(ret));
361 return false;
362 }
363
364 // Step 3: Create and cache device handle for the BNO08x address
365 // Note: scl_wait_us sets maximum clock stretching timeout per byte
366 // BNO08x can stretch clock when it has no data, so we allow up to 50ms
367 // This must be long enough for the device to prepare data or timeout gracefully
368 i2c_device_config_t dev_config = {
369 .dev_addr_length = I2C_ADDR_BIT_LEN_7,
370 .device_address = config_.device_address,
371 .scl_speed_hz = config_.frequency,
372 .scl_wait_us = 50000, // 50ms max clock stretching per byte (BNO08x datasheet requirement)
373 .flags = {},
374 };
375
376 ret = i2c_master_bus_add_device(bus_handle_, &dev_config, &dev_handle_);
377 if (ret != ESP_OK) {
378 ESP_LOGE(TAG_I2C, "Failed to add device 0x%02X: %s",
379 config_.device_address, esp_err_to_name(ret));
380 i2c_del_master_bus(bus_handle_);
381 bus_handle_ = nullptr;
382 return false;
383 }
384
385 initialized_ = true;
386 ESP_LOGI(TAG_I2C, "I2C bus initialized successfully");
387 return true;
388 }

448 void HardwareReset(uint32_t lowMs = 2, uint32_t bootDelayMs = 200) noexcept {
449 if (config_.rst_pin == GPIO_NUM_NC) {
450 ESP_LOGW(TAG_I2C, "Reset pin not configured, skipping hardware reset");
451 return;
452 }
453
454 ESP_LOGI(TAG_I2C, "Performing hardware reset: asserting RSTN for %lu ms",
455 static_cast<unsigned long>(lowMs));
456
457 // Assert reset (drive LOW)
458 SetReset(true);
459 Delay(lowMs);
460
461 // Release reset (drive HIGH)
462 SetReset(false);
463 ESP_LOGI(TAG_I2C, "Reset released, waiting %lu ms for sensor boot",
464 static_cast<unsigned long>(bootDelayMs));
465 Delay(bootDelayMs);
466 }

478 bool initGPIO() noexcept {
479 // Configure interrupt pin (input, pull-up enabled, active low)
480 if (config_.int_pin != GPIO_NUM_NC) {
481 gpio_config_t io_conf = {};
482 io_conf.pin_bit_mask = (1ULL << config_.int_pin);
483 io_conf.mode = GPIO_MODE_INPUT;
484 io_conf.pull_up_en = GPIO_PULLUP_ENABLE;
485 io_conf.pull_down_en = GPIO_PULLDOWN_DISABLE;
486 io_conf.intr_type = GPIO_INTR_DISABLE;
487
488 esp_err_t ret = gpio_config(&io_conf);
489 if (ret != ESP_OK) {
490 ESP_LOGE(TAG_I2C, "Failed to configure interrupt pin: %s", esp_err_to_name(ret));
491 return false;
492 }
493 ESP_LOGI(TAG_I2C, "Interrupt pin configured: GPIO%d", config_.int_pin);
494 }
495
496 // Configure reset pin (output, high by default = not asserted)
497 if (config_.rst_pin != GPIO_NUM_NC) {
498 gpio_config_t io_conf = {};
499 io_conf.pin_bit_mask = (1ULL << config_.rst_pin);
500 io_conf.mode = GPIO_MODE_OUTPUT;
501 io_conf.pull_up_en = GPIO_PULLUP_DISABLE;
502 io_conf.pull_down_en = GPIO_PULLDOWN_DISABLE;
503 io_conf.intr_type = GPIO_INTR_DISABLE;
504
505 esp_err_t ret = gpio_config(&io_conf);
506 if (ret != ESP_OK) {
507 ESP_LOGE(TAG_I2C, "Failed to configure reset pin: %s", esp_err_to_name(ret));
508 return false;
509 }
510
511 // Release reset (set high)
512 gpio_set_level(config_.rst_pin, 1);
513 ESP_LOGI(TAG_I2C, "Reset pin configured: GPIO%d", config_.rst_pin);
514 }
515
516 return true;
517 }
