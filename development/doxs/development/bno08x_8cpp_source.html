54template <typename CommType>
55bool BNO085<CommType>::Begin() noexcept {
56 if (state_ == BNO085DriverState::Sh2Active)
57 return true;
58 if (state_ != BNO085DriverState::Closed) {
59 last_error_ = SH2_ERR_OP_IN_PROGRESS;
60 return false;
61 }
62 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC) {
63 last_error_ = SH2_ERR_BAD_PARAM;
64 return false;
65 }
66
67 last_error_ = 0;
68 prepareHalWrapper();
69
70 int status = sh2_open(halWrapper_.asHal(), asyncCallback, this);
71 if (status != SH2_OK) {
72 last_error_ = status;
73 io_.Close();
74 return false;
75 }
76 status = sh2_reinitialize();
77 if (status != SH2_OK) {
78 last_error_ = status;
79 sh2_close();
80 io_.Close();
81 return false;
82 }
83 status = sh2_setSensorCallback(sensorCallback, this);
84 if (status != SH2_OK) {
85 last_error_ = status;
86 sh2_close();
87 io_.Close();
88 return false;
89 }
90
91 new_flag_.fill(false);
92 last_interval_.fill(0);
93 last_sensitivity_.fill(0);
94 state_ = BNO085DriverState::Sh2Active;
95 return true;
96}

105template <typename CommType>
106bool BNO085<CommType>::EnableSensor(BNO085Sensor sensor, uint32_t interval_ms,
107 float sensitivity) noexcept {
108 if (state_ != BNO085DriverState::Sh2Active) {
109 last_error_ = SH2_ERR_OP_IN_PROGRESS;
110 return false;
111 }
112 auto id = static_cast<uint8_t>(sensor);
113 if (id >= last_interval_.size())
114 return false;
115 uint32_t interval_us = interval_ms * 1000;
116 if (!configure(sensor, interval_us, sensitivity, 0))
117 return false;
118 last_interval_[id] = interval_us;
119 last_sensitivity_[id] = sensitivity;
120 return true;
121}

527template <typename CommType>
528void BNO085<CommType>::handleSensorEvent(const sh2_SensorEvent_t* event) noexcept {
529 sh2_SensorValue_t value;
530 int decode_status = sh2_decodeSensorEvent(&value, event);
531 if (decode_status != SH2_OK) {
532 last_error_ = decode_status;
533 return;
534 }
535 uint8_t id = value.sensorId;
536 if (id >= latest_.size())
537 return;
538 latest_[id] = value;
539 new_flag_[id] = true;
540 if (callback_) {
541 callback_(toSensorEvent(static_cast<BNO085Sensor>(id), value));
542 }
543}

551template <typename CommType>
552void BNO085<CommType>::handleAsyncEvent(const sh2_AsyncEvent_t* event) noexcept {
553 if (event->eventId == SH2_RESET) {
554 for (uint8_t id = 0; id < last_interval_.size(); ++id) {
555 if (last_interval_[id])
556 configure(static_cast<BNO085Sensor>(id), last_interval_[id], last_sensitivity_[id], 0);
557 }
558 }

570template <typename CommType>
571bool BNO085<CommType>::configure(BNO085Sensor sensor, uint32_t interval_us, float sensitivity,
572 uint32_t batch_us) noexcept {
573 sh2_SensorConfig_t cfg{};
574 cfg.reportInterval_us = interval_us;
575 cfg.batchInterval_us = batch_us;
576 cfg.sensorSpecific = 0;
577 const float clamped_sensitivity = std::clamp(sensitivity, 0.0f, 65535.0f);
578 cfg.changeSensitivity = static_cast<uint16_t>(clamped_sensitivity + 0.5f);
579 cfg.changeSensitivityEnabled = clamped_sensitivity > 0.0f;
580 int status = sh2_setSensorConfig(static_cast<sh2_SensorId_t>(sensor), &cfg);
581 if (status != SH2_OK) {
582 last_error_ = status;
583 return false;
584 }
585 return true;
586}

614template <typename CommType>
615bool BNO085<CommType>::EnterBootloader(uint32_t resetLowMs, uint32_t settleMs) noexcept {
616 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC) {
617 last_error_ = SH2_ERR_BAD_PARAM;
618 return false;
619 }
620 if (state_ == BNO085DriverState::DfuInProgress) {
621 last_error_ = SH2_ERR_OP_IN_PROGRESS;
622 return false;
623 }
624 if (state_ != BNO085DriverState::Closed) {
625 Close();
626 }
627 if (state_ != BNO085DriverState::Closed) {
628 last_error_ = SH2_ERR_OP_IN_PROGRESS;
629 return false;
630 }
631
632 io_.SetBoot(true); // BOOTN active-low: true drives pin low.
633 HardwareReset(resetLowMs);
634 io_.SetBoot(false); // Release BOOTN high after reset.
635 if (settleMs) {
636 io_.Delay(settleMs);
637 }
638 last_error_ = SH2_OK;
639 return true;
640}

643template <typename CommType>

645 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC) {
646 last_error_ = SH2_ERR_BAD_PARAM;
647 return false;
648 }
649 if (state_ == BNO085DriverState::DfuInProgress) {

656 if (state_ != BNO085DriverState::Closed) {

662 HardwareReset(resetLowMs);
663 if (settleMs) {
664 io_.Delay(settleMs);
665 }
666 last_error_ = SH2_OK;

789template <typename CommType>
790void BNO085<CommType>::rvcProcessByte(uint8_t c) noexcept {
791 if (rvc_frame_len_ == RVC_FRAME_LEN_) {
792 std::memmove(rvc_frame_, rvc_frame_ + 1, RVC_FRAME_LEN_ - 1);
793 rvc_frame_[RVC_FRAME_LEN_ - 1] = c;
794 } else {
795 rvc_frame_[rvc_frame_len_++] = c;
796 }
797
798 if (rvc_frame_len_ == RVC_FRAME_LEN_ && rvc_frame_[0] == 0xAA && rvc_frame_[1] == 0xAA) {
799 uint8_t check = 0;
800 for (int i = 2; i < RVC_FRAME_LEN_ - 1; ++i)
801 check += rvc_frame_[i];
802
803 if (check == rvc_frame_[RVC_FRAME_LEN_ - 1]) {
804 RvcSensorEvent event{};
805 event.timestamp_uS = io_.GetTimeUs();
806 event.index = rvc_frame_[2];
807 event.yaw = static_cast<int16_t>((rvc_frame_[4] << 8) | rvc_frame_[3]);
808 event.pitch = static_cast<int16_t>((rvc_frame_[6] << 8) | rvc_frame_[5]);
809 event.roll = static_cast<int16_t>((rvc_frame_[8] << 8) | rvc_frame_[7]);
810 event.acc_x = static_cast<int16_t>((rvc_frame_[10] << 8) | rvc_frame_[9]);
811 event.acc_y = static_cast<int16_t>((rvc_frame_[12] << 8) | rvc_frame_[11]);
812 event.acc_z = static_cast<int16_t>((rvc_frame_[14] << 8) | rvc_frame_[13]);
813 event.mi = rvc_frame_[15];
814 event.mr = rvc_frame_[16];
815
816 if (rvc_cb_) {
817 RvcSensorValue val;
818 decodeRvc(&val, &event);
819 rvc_cb_(val);
820 }
821 rvc_frame_len_ = 0;
822 }
823 }

888template <typename CommType>
889void BNO085<CommType>::dfuAppendCrc(uint8_t* packet, uint8_t len) noexcept {
890 uint16_t crc = 0xFFFF;
891 for (int n = 0; n < len; n++) {
892 uint16_t x = static_cast<uint16_t>(packet[n]) << 8;
893 for (int i = 0; i < 8; i++) {
894 if ((crc ^ x) & 0x8000)
895 crc = (crc << 1) ^ 0x1021;
896 else
897 crc = crc << 1;
898 x <<= 1;
899 }
900 }
901 packet[len] = (crc >> 8) & 0xFF;
902 packet[len + 1] = crc & 0xFF;
903}

913template <typename CommType>
914int BNO085<CommType>::dfuSend(uint8_t* dfu_buff, uint8_t* p_data, uint32_t len) noexcept {
915 unsigned int retries = 0;
916 int status = SH2_OK;
917 uint8_t ack = 0;
918 bool got_ack = false;
919 uint32_t t;
920 sh2_Hal_t* hal = halWrapper_.asHal();
921
922 while (!got_ack && (retries < DFU_MAX_ATTEMPTS)) {
923 uint32_t now = hal->getTimeUs(hal);
924 uint32_t start = now;
925 status = 0;
926 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
927 status = hal->write(hal, p_data, len);
928 now = hal->getTimeUs(hal);
929 }
930 if (status == 0)
931 status = SH2_ERR_TIMEOUT;
932 if (status > 0) {
933 status = 0;
934 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
935 status = hal->read(hal, &ack, 1, &t);
936 now = hal->getTimeUs(hal);
937 }
938 if (status == 0)
939 status = SH2_ERR_TIMEOUT;
940 }
941 if (status > 0) {
942 if (ack == DFU_ACK) {
943 got_ack = true;
944 status = SH2_OK;
945 } else {
946 got_ack = false;
947 status = SH2_ERR_HUB;
948 }
949 }
950 if (!got_ack)
951 retries++;
952 }
953 return (status >= 0) ? SH2_OK : status;
954}

1039template <typename CommType>
1040int BNO085<CommType>::DfuWithOptions(const HcBin_t& fw, const DfuOptions& options) noexcept {
1041 if (io_.GetInterfaceType() == BNO085Interface::UARTRVC) {
1042 last_error_ = SH2_ERR_BAD_PARAM;
1043 return SH2_ERR;
1044 }
1045 if (state_ == BNO085DriverState::RvcActive || state_ == BNO085DriverState::DfuInProgress) {
1046 last_error_ = SH2_ERR_OP_IN_PROGRESS;
1047 return SH2_ERR_OP_IN_PROGRESS;
1048 }
1049 if (state_ == BNO085DriverState::Sh2Active) {
1050 // DFU uses the same transport HAL and must not race an active SH-2 session.
1051 Close();
1052 }
1053 if (state_ != BNO085DriverState::Closed) {
1054 last_error_ = SH2_ERR_OP_IN_PROGRESS;
1055 return SH2_ERR_OP_IN_PROGRESS;
1056 }
1057 state_ = BNO085DriverState::DfuInProgress;
1058 prepareHalWrapper();
1059
1060 int rc, status = SH2_OK;
1061 uint32_t app_len = 0;
1062 uint8_t packet_len = 0;
1063 uint32_t offset = 0;
1064 const char* meta = nullptr;
1065 bool fw_opened = false;
1066 bool hal_opened = false;
1067 uint8_t dfu_buff[DFU_MAX_PACKET_LEN + 2];
1068 sh2_Hal_t* hal = halWrapper_.asHal();
1069
1070 rc = fw.open();
1071 if (rc != 0) {
1072 status = SH2_ERR;
1073 goto dfu_end;
1074 }
1075 fw_opened = true;
1076
1077 if (options.requireFormatMatch) {
1078 const char* expected_format = options.requiredFormat ? options.requiredFormat : "BNO_V1";
1079 meta = fw.getMeta("FW-Format");
1080 if (!meta || std::strcmp(meta, expected_format) != 0) {
1081 status = SH2_ERR_BAD_PARAM;
1082 goto dfu_close;
1083 }
1084 }
1085
1086 if (options.requirePartNumber) {
1087 meta = fw.getMeta("SW-Part-Number");
1088 if (!meta) {
1089 status = SH2_ERR_BAD_PARAM;
1090 goto dfu_close;
1091 }
1092 if (options.requiredPartNumber) {
1093 if (std::strcmp(meta, options.requiredPartNumber) != 0) {
1094 status = SH2_ERR_BAD_PARAM;
1095 goto dfu_close;
1096 }
1097 } else if (!dfuIsKnownPartNumber(meta)) {
1098 status = SH2_ERR_BAD_PARAM;
1099 goto dfu_close;
1100 }
1101 }
1102
1103 app_len = fw.getAppLen();
1104 if (app_len < 1024) {
1105 status = SH2_ERR_BAD_PARAM;
1106 goto dfu_close;
1107 }
1108
1109 packet_len = fw.getPacketLen();
1110 if (options.packetLenOverride != 0) {
1111 packet_len = options.packetLenOverride;
1112 }
1113 if (packet_len == 0 || packet_len > DFU_MAX_PACKET_LEN)
1114 packet_len = DFU_MAX_PACKET_LEN;
1115
1116 status = hal->open(hal);
1117 if (status != SH2_OK)
1118 goto dfu_close;
1119 hal_opened = true;
1120
1121 status = dfuSendAppSize(dfu_buff, app_len);
1122 if (status != SH2_OK)
1123 goto dfu_close;
1124 status = dfuSendPktSize(dfu_buff, packet_len);
1125 if (status != SH2_OK)
1126 goto dfu_close;
1127
1128 if (options.progress) {
1129 options.progress(DfuProgress{0, app_len});
1130 }
1131
1132 offset = 0;
1133 while (offset < app_len) {
1134 uint32_t to_send = app_len - offset;
1135 if (to_send > packet_len)
1136 to_send = packet_len;
1137 status = fw.getAppData(dfu_buff, offset, to_send);
1138 if (status != SH2_OK) {
1139 status = SH2_ERR;
1140 goto dfu_close;
1141 }
1142 status = dfuSendPkt(dfu_buff, dfu_buff, to_send);
1143 if (status != SH2_OK)
1144 goto dfu_close;
1145 offset += to_send;
1146 if (options.progress) {
1147 options.progress(DfuProgress{offset, app_len});
1148 }
1149 }
1150
1151dfu_close:
1152 if (fw_opened)
1153 fw.close();
1154 if (status == SH2_OK && hal_opened) {
1155 uint32_t now = hal->getTimeUs(hal), start = now;
1156 while ((now - start) < DFU_DELAY_POST_US)
1157 now = hal->getTimeUs(hal);
1158 }
1159 if (hal_opened)
1160 hal->close(hal);
1161
1162dfu_end:
1163 state_ = BNO085DriverState::Closed;
1164 last_error_ = status;
1165 return status;
1166}
