103template <typename CommType>
104SensorEvent BNO085<CommType>::GetLatest(BNO085Sensor sensor) const {
105 SensorEvent out{};
106 out.sensor = sensor;
107 auto id = static_cast<uint8_t>(sensor);
108 const auto& val = latest_[id];
109 out.timestamp = val.timestamp;
110 uint8_t accuracy = val.status & 0x03;
111 switch (sensor) {
112 case BNO085Sensor::Accelerometer:
113 case BNO085Sensor::LinearAcceleration:
114 case BNO085Sensor::Gravity:
115 out.vector.x = val.un.accelerometer.x;
116 out.vector.y = val.un.accelerometer.y;
117 out.vector.z = val.un.accelerometer.z;
118 out.vector.accuracy = accuracy;
119 break;
120 case BNO085Sensor::Gyroscope:
121 out.vector.x = val.un.gyroscope.x;
122 out.vector.y = val.un.gyroscope.y;
123 out.vector.z = val.un.gyroscope.z;
124 out.vector.accuracy = accuracy;
125 break;
126 case BNO085Sensor::Magnetometer:
127 out.vector.x = val.un.magneticField.x;
128 out.vector.y = val.un.magneticField.y;
129 out.vector.z = val.un.magneticField.z;
130 out.vector.accuracy = accuracy;
131 break;
132 case BNO085Sensor::RotationVector:
133 case BNO085Sensor::GameRotationVector:
134 case BNO085Sensor::GeomagneticRotationVector:
135 case BNO085Sensor::ARVRStabilizedRV:
136 case BNO085Sensor::ARVRStabilizedGameRV:
137 out.rotation.w = val.un.rotationVector.real;
138 out.rotation.x = val.un.rotationVector.i;
139 out.rotation.y = val.un.rotationVector.j;
140 out.rotation.z = val.un.rotationVector.k;
141 out.rotation.accuracy = accuracy;
142 break;
143 case BNO085Sensor::GyroIntegratedRV:
144 out.rotation.w = val.un.gyroIntegratedRV.real;
145 out.rotation.x = val.un.gyroIntegratedRV.i;
146 out.rotation.y = val.un.gyroIntegratedRV.j;
147 out.rotation.z = val.un.gyroIntegratedRV.k;
148 break;
149 case BNO085Sensor::StepCounter:
150 out.stepCount = val.un.stepCounter.steps;
151 break;
152 case BNO085Sensor::TapDetector:
153 out.tap.doubleTap = (val.un.tapDetector.flags & TAPDET_DOUBLE);
154 if (val.un.tapDetector.flags & TAPDET_X) {
155 out.tap.direction = (val.un.tapDetector.flags & TAPDET_X_POS) ? 0 : 1;
156 } else if (val.un.tapDetector.flags & TAPDET_Y) {
157 out.tap.direction = (val.un.tapDetector.flags & TAPDET_Y_POS) ? 2 : 3;
158 } else if (val.un.tapDetector.flags & TAPDET_Z) {
159 out.tap.direction = (val.un.tapDetector.flags & TAPDET_Z_POS) ? 4 : 5;
160 } else {
161 out.tap.direction = 0;
162 }
163 out.detected = val.un.tapDetector.flags & (TAPDET_X | TAPDET_Y | TAPDET_Z);
164 break;
165 default:
166 break;
167 }
168 return out;
169}

250template <typename CommType>
251void BNO085<CommType>::handleSensorEvent(const sh2_SensorEvent_t* event) {
252 sh2_SensorValue_t value;
253 sh2_decodeSensorEvent(&value, event);
254 uint8_t id = value.sensorId;
255 if (id >= latest_.size())
256 return;
257 latest_[id] = value;
258 new_flag_[id] = true;
259 if (callback_) {
260 callback_(GetLatest(static_cast<BNO085Sensor>(id)));
261 new_flag_[id] = false;
262 }
263}

268 if (event->eventId == SH2_RESET) {

270 if (last_interval_[id]) {

272 }
