44static inline void dfu_write32be(uint8_t* buf, uint32_t value) {
45 *buf++ = (value >> 24) & 0xFF;
46 *buf++ = (value >> 16) & 0xFF;
47 *buf++ = (value >> 8) & 0xFF;
48 *buf++ = (value >> 0) & 0xFF;
49}

51static inline void dfu_appendCrc(uint8_t* packet, uint8_t len) {
52 uint16_t crc;
53 uint16_t x;
54
55 // compute CRC of packet
56 crc = 0xFFFF;
57 for (int n = 0; n < len; n++) {
58 x = (uint16_t)(packet[n]) << 8;
59 for (int i = 0; i < 8; i++) {
60 if ((crc ^ x) & 0x8000) {
61 crc = (crc << 1) ^ 0x1021;
62 } else {
63 crc = crc << 1;
64 }
65 x <<= 1;
66 }
67 }
68
69 // Append the CRC to packet
70 packet[len] = (crc >> 8) & 0xFF;
71 packet[len + 1] = crc & 0xFF;
72}

76template <typename TransportType>
77static int dfuSend(TransportType* pHal, uint8_t* dfuBuff, uint8_t* pData, uint32_t len) {
78 unsigned int retries = 0;
79 int status = SH2_OK;
80 uint8_t ack = 0;
81 bool gotAck = false;
82 uint32_t t;
83
84 while (!gotAck && (retries < DFU_MAX_ATTEMPTS)) {
85 uint32_t now = pHal->GetTimeUs();
86 uint32_t start = now;
87
88 // Do write
89 status = 0;
90 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
91 status = pHal->Write(pData, len);
92 now = pHal->GetTimeUs();
93 }
94 if (status == 0) {
95 // recognize timeout as an error.
96 status = SH2_ERR_TIMEOUT;
97 }
98
99 // If write succeeded, read ack
100 if (status > 0) {
101 status = 0;
102 while ((status == 0) && ((now - start) < DFU_SEND_TIMEOUT_US)) {
103 status = pHal->Read(&ack, 1, &t);
104 now = pHal->GetTimeUs();
105 }
106 if (status == 0) {
107 // Recognize timeout as an error
108 status = SH2_ERR_TIMEOUT;
109 }
110 }
111
112 // If read succeeded, check for ACK
113 if (status > 0) {
114 if (ack == ACK) {
115 // We got a good ack
116 gotAck = true;
117 status = SH2_OK;
118 } else {
119 // We got NAK
120 gotAck = false;
121 status = SH2_ERR_HUB;
122 }
123 }
124
125 if (!gotAck) {
126 // Problem: try again.
127 retries++;
128 }
129 }
130
131 if (status >= 0) {
132 status = SH2_OK;
133 }
134
135 return status;
136}

138template <typename TransportType>
139static int dfu_sendAppSize(TransportType* pHal, uint8_t* dfuBuff, uint32_t appSize) {
140 dfu_write32be(dfuBuff, appSize);
141 dfu_appendCrc(dfuBuff, 4);
142 return dfuSend(pHal, dfuBuff, dfuBuff, 6);
143}
144
145template <typename TransportType>
146static int dfu_sendPktSize(TransportType* pHal, uint8_t* dfuBuff, uint8_t packetLen) {
147 dfuBuff[0] = packetLen;
148 dfu_appendCrc(dfuBuff, 1);
149 return dfuSend(pHal, dfuBuff, dfuBuff, 3);
150}
151
152template <typename TransportType>
153static int dfu_sendPkt(TransportType* pHal, uint8_t* dfuBuff, uint8_t* pData, uint32_t len) {
154 memcpy(dfuBuff, pData, len);
155 dfu_appendCrc(dfuBuff, len);
156 return dfuSend(pHal, dfuBuff, dfuBuff, len + 2); // plus 2 for CRC
157}

161template <typename TransportType>
162int dfu(TransportType& transport, const HcBin_t& fw) {
163 int rc;
164 int status = SH2_OK;
165 uint32_t appLen = 0;
166 uint8_t packetLen = 0;
167 uint32_t offset = 0;
168 const char* s = 0;
169 TransportType* pHal = &transport;
170 uint8_t dfuBuff[MAX_PACKET_LEN + 2];
171
172 // Open the hcbin object
173 rc = fw.open();
174 if (rc != 0) {
175 status = SH2_ERR;
176 goto end;
177 }
178
179 // Validate firmware matches this implementation
180 s = fw.getMeta("FW-Format");
181 if ((s == 0) || (strcmp(s, "BNO_V1") != 0)) {
182 // No format info or Incorrect format
183 status = SH2_ERR_BAD_PARAM;
184 goto close_and_return;
185 }
186
187 // Validate firmware is for the right part number
188 s = fw.getMeta("SW-Part-Number");
189 if (s == 0) {
190 // No part number info
191 status = SH2_ERR_BAD_PARAM;
192 goto close_and_return;
193 }
194 if ((strcmp(s, "1000-3608") != 0) && (strcmp(s, "1000-3676") != 0) &&
195 (strcmp(s, "1000-4148") != 0) && (strcmp(s, "1000-4563") != 0)) {
196 // Incorrect part number
197 status = SH2_ERR_BAD_PARAM;
198 goto close_and_return;
199 }
200
201 // Validate firmware length
202 appLen = fw.getAppLen();
203 if (appLen < 1024) {
204 // App data is empty
205 status = SH2_ERR_BAD_PARAM;
206 goto close_and_return;
207 }
208
209 // Determine packet length to use
210 packetLen = fw.getPacketLen();
211 if ((packetLen == 0) || (packetLen > MAX_PACKET_LEN)) {
212 packetLen = MAX_PACKET_LEN;
213 }
214
215 // Initiate DFU process
216
217 status = pHal->Open();
218 if (status != SH2_OK) {
219 goto close_and_return;
220 }
221
222 // Send app size
223 status = dfu_sendAppSize(pHal, dfuBuff, appLen);
224 if (status != SH2_OK) {
225 goto close_and_return;
226 }
227
228 // Send packet size
229 status = dfu_sendPktSize(pHal, dfuBuff, packetLen);
230 if (status != SH2_OK) {
231 goto close_and_return;
232 }
233
234 // Send firmware image
235 offset = 0;
236 while (offset < appLen) {
237 uint32_t toSend = appLen - offset;
238 if (toSend > packetLen) {
239 toSend = packetLen;
240 }
241
242 // Extract this packet's content from hcbin
243 status = fw.getAppData(dfuBuff, offset, toSend);
244 if (status != SH2_OK) {
245 goto close_and_return;
246 }
247
248 // Send this packet's contents
249 status = dfu_sendPkt(pHal, dfuBuff, dfuBuff, toSend);
250 if (status != SH2_OK) {
251 goto close_and_return;
252 }
253
254 // update loop variables
255 offset += toSend;
256 }
257
258close_and_return:
259 // close firmware
260 fw.close();
261
262 // If update process completed successfully, delay a bit to let
263 // flash writes complete.
264 if (status == SH2_OK) {
265 uint32_t now = pHal->GetTimeUs();
266 uint32_t start = now;
267 while ((now - start) < DELAY_POST_DFU_US) {
268 now = pHal->GetTimeUs();
269 }
270 }
271
272 // close device
273 pHal->Close();
274
275end:
276 return status;
277}
